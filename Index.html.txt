// ===================== STATE =====================
let state = {
  currentWeek: 1,
  currentDay: "DOS + ÉPAULES",
  logs: {},
  morpho: [],
  pullups: [],
  dips: [],
  pushups: []
};

function loadState() {
  try {
    const s = localStorage.getItem('eliteCoachState');
    if (s) { const parsed = JSON.parse(s); if (parsed && typeof parsed === 'object') state = parsed; }
  } catch(e) { console.log('load error:', e); }
}

function saveState() {
  try {
    const json = JSON.stringify(state);
    localStorage.setItem('eliteCoachState', json);
  } catch(e) {
    console.log('save error:', e);
  }
}

// ===================== NAV =====================
function showPage(id, btn) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
  var pg = document.getElementById('page-' + id);
  if (pg) pg.classList.add('active');
  if (btn) btn.classList.add('active');
  else { var found = document.querySelector('.nb-' + id); if(found) found.classList.add('active'); }
}

// ===================== WEEK GRID =====================
function renderWeekGrid() {
  const grid = document.getElementById('weekGrid');
  if (!grid) return;
  grid.innerHTML = '';
  for (let w = 1; w <= 12; w++) {
    const phase = program.phases.find(p => p.weeks.includes(w));
    const btn = document.createElement('div');
    btn.className = 'week-btn' + (w === state.currentWeek ? ' active' : '');
    btn.innerHTML = `S${w}<div class="phase-tag" style="color:${phase.color}; font-size:8px; margin-top:2px">${phase.name.substring(0,4)}</div>`;
    btn.onclick = () => selectWeek(w);
    grid.appendChild(btn);
  }
}

function selectWeek(w) {
  state.currentWeek = w;
  saveState();
  renderWeekGrid();
  renderDayRow();
  renderWorkout();
  document.getElementById('headerWeek').textContent = `S${w} — ${state.currentDay.split(' ')[0]}`;
}

// ===================== DAY ROW =====================
function renderDayRow() {
  const row = document.getElementById('dayRow');
  if (!row) return;
  row.innerHTML = '';
  dayOrder.forEach(day => {
    const btn = document.createElement('button');
    btn.className = 'day-btn' + (day === state.currentDay ? ' active' : '');
    btn.textContent = day;
    btn.onclick = () => selectDay(day);
    row.appendChild(btn);
  });
}

function selectDay(day) {
  state.currentDay = day;
  saveState();
  renderDayRow();
  renderWorkout();
}

// ===================== WORKOUT RENDER =====================
function renderWorkout() {
  const container = document.getElementById('workoutContent');
  if (!container) return;
  const session = program.sessions[state.currentDay];
  if (!session) return;
  const mod = weekMods[state.currentWeek];
  const phase = program.phases.find(p => p.weeks.includes(state.currentWeek));

  let html = `
    <div class="card">
      <div class="card-title">
        <span class="icon">💪</span> S${state.currentWeek} — ${state.currentDay}
        <span class="badge" style="background:${phase.color};color:#000;margin-left:auto">${phase.name}</span>
      </div>
      <p style="margin-bottom:10px">${session.desc}</p>
      <div class="alert info"><strong>Note semaine :</strong> ${mod.note}</div>
      <div style="font-size:12px; color:var(--text-muted); margin-bottom:14px">
        <strong style="color:var(--accent2)">🔥 WARM-UP :</strong> ${session.warmup}
      </div>
  `;

  session.exercises.forEach((ex, idx) => {
    const logKey = `S${state.currentWeek}_${state.currentDay}_ex${idx}`;
    const log = state.logs[logKey] || {};
    const setsNum = parseInt(ex.sets) + (idx === 0 ? mod.sets : 0);
    
    html += `
      <div style="padding: 12px 0; border-bottom: 1px solid var(--border);">
        <div style="display:flex; align-items:flex-start; gap:10px; flex-wrap:wrap; margin-bottom:8px">
          <div style="flex:1; min-width:180px">
            <div class="ex-name" style="font-size:14px">${ex.name}</div>
            <div class="ex-note">${ex.note}</div>
          </div>
          <div style="text-align:right; flex-shrink:0">
            <div style="font-family:'Courier New','Menlo',monospace; font-size:13px; color:var(--accent)">${setsNum}×${ex.reps}</div>
            ${ex.tempo ? `<div style="font-size:10px;color:var(--text-muted)">Tempo: ${ex.tempo}</div>` : ''}
            <div style="font-size:10px;color:var(--text-muted)">Repos: ${ex.rest}</div>
          </div>
        </div>
        <div class="input-row" style="flex-wrap:wrap">
          <div>
            <div class="mini-label">Reps</div>
            <input class="mini-input" type="number" min="0" max="100" value="${log.reps || ''}" placeholder="—" id="reps_${logKey}" style="width:60px">
          </div>
          <div>
            <div class="mini-label">Charge (kg)</div>
            <input class="mini-input" type="number" min="0" max="300" step="0.5" value="${log.weight || ''}" placeholder="BW" id="weight_${logKey}" style="width:70px">
          </div>
          <div>
            <div class="mini-label">Séries OK</div>
            <input class="mini-input" type="number" min="0" max="10" value="${log.sets || ''}" placeholder="${setsNum}" id="sets_${logKey}" style="width:60px">
          </div>
          <button class="save-btn ${log.saved ? 'saved' : ''}" onclick="logExercise('${logKey}')" id="savebtn_${logKey}">
            ${log.saved ? '✓ Sauvé' : 'Sauver'}
          </button>
        </div>
        ${log.saved ? `<div style="font-size:11px; color:var(--success); margin-top:6px">✓ ${log.reps} reps ${log.weight ? '@ '+log.weight+'kg' : '(poids corps)'} × ${log.sets || setsNum} séries</div>` : ''}
      </div>
    `;
  });

  html += `</div>`;
  container.innerHTML = html;
}

function logExercise(key) {
  const reps = document.getElementById('reps_' + key)?.value;
  const weight = document.getElementById('weight_' + key)?.value;
  const sets = document.getElementById('sets_' + key)?.value;
  state.logs[key] = { reps, weight, sets, saved: true, date: new Date().toLocaleDateString('fr') };
  saveState();
  const btn = document.getElementById('savebtn_' + key);
  if (btn) { btn.textContent = '✓ Sauvé'; btn.classList.add('saved'); }
  renderWorkout();
}

// ===================== WEEK PROGRESS =====================
function renderWeekProgress() {
  const container = document.getElementById('weekProgressBars');
  if (!container) return;
  let html = '';
  program.phases.forEach(phase => {
    html += `<div style="margin-bottom:14px">`;
    html += `<div style="font-size:11px; letter-spacing:2px; text-transform:uppercase; color:${phase.color}; margin-bottom:6px">${phase.name}</div>`;
    phase.weeks.forEach(w => {
      const logsCount = Object.keys(state.logs).filter(k => k.startsWith(`S${w}_`)).length;

      html += `

          <div style="width:24px; font-family:'Courier New','Menlo',monospace; font-size:11px; color:var(--text-muted)">S${w}</div>
          <div style="flex:1">
            <div class="prog-bar-wrap">
              <div class="prog-bar" style="width:${pct}%; background:${phase.color}"></div>
            </div>
          </div>
          <div style="width:30px; font-family:'Courier New','Menlo',monospace; font-size:10px; color:${phase.color}; text-align:right">${pct}%</div>
        </div>
      `;
    });
    html += `</div>`;
  });
  container.innerHTML = html;
}

        <div style="font-size:10px; color:var(--text-muted); text-transform:uppercase; letter-spacing:1px">Départ</div>

function logPerf(key) {
  const val = parseInt(document.getElementById('perf_' + key)?.value);
  if (isNaN(val)) return;
  if (!state[key]) state[key] = [];
  state[key].push({ val, date: new Date().toLocaleDateString('fr') });
  saveState();
  renderPerformances();
}

  renderMorpho();
}

      <div style="display:flex; gap:14px; flex-wrap:wrap">
        ${d.weight ? `<span>${d.weight}kg</span>` : ''}
        ${d.waist ? `<span>Taille: ${d.waist}cm</span>` : ''}
        ${d.chest ? `<span>Pecs: ${d.chest}cm</span>` : ''}
        ${d.arm ? `<span>Bras: ${d.arm}cm</span>` : ''}
      </div>
    </div>`;
  });
  html += `</div>`;
  c.innerHTML = html;
}

function logMorpho() {
  const entry = {
    date: new Date().toLocaleDateString('fr'),
    weight: document.getElementById('morph_weight')?.value,
    waist: document.getElementById('morph_waist')?.value,
    chest: document.getElementById('morph_chest')?.value,
    arm: document.getElementById('morph_arm')?.value,
  };
  if (!state.morpho) state.morpho = [];
  state.morpho.push(entry);
  // update dashboard stat
  if (entry.weight) document.getElementById('dash-weight').textContent = entry.weight;
  saveState();
  renderMorpho();
}

// ===================== TIMER =====================
let timerVal = 90, timerRunning = false, timerInterval = null;

function formatTime(s) {
  const m = Math.floor(s / 60).toString().padStart(2, '0');
  const sec = (s % 60).toString().padStart(2, '0');
  return m + ':' + sec;
}

function setTimer(s) { 
  pauseTimer();
  timerVal = s;
  document.getElementById('timerDisplay').textContent = formatTime(s);
}

function startTimer() {
  if (timerRunning) return;
  timerRunning = true;
  timerInterval = setInterval(() => {
    timerVal--;
    document.getElementById('timerDisplay').textContent = formatTime(timerVal);
    if (timerVal <= 0) {
      pauseTimer();
      document.getElementById('timerDisplay').textContent = 'GO ! 💪';
    }
  }, 1000);
}

function pauseTimer() {
  timerRunning = false;
  clearInterval(timerInterval);
}

function resetTimer() {
  pauseTimer();
  timerVal = 90;
  document.getElementById('timerDisplay').textContent = formatTime(90);
}

// ===================== SESSIONS COUNTER =====================
function updateSessionsCount() {
  const thisWeek = Object.keys(state.logs).filter(k => k.startsWith(`S${state.currentWeek}_`));
  const days = new Set(thisWeek.map(k => k.split('_')[1]));
  const el = document.getElementById('dash-sessions');
  if (el) el.textContent = days.size;
}

}
